[binaries]
c = 'x86_64-w64-mingw32-gcc'
cpp = 'x86_64-w64-mingw32-g++'
ar = 'x86_64-w64-mingw32-ar'
strip = 'x86_64-w64-mingw32-strip'
windres = 'x86_64-w64-mingw32-windres'

[properties]
needs_exe_wrapper = true

[host_machine]
system = 'windows'
cpu_family = 'x86_64'
cpu = 'x86_64'
endian = 'little'

# The same semantics the release is built with (see .github/crossfiles).
# meson's release build type adds -O3 and neither of these, so a local build
# and an Actions artifact could optimise differently from the payload Autorun
# installs -- and then a crash is impossible to reproduce across the two.
[built-in options]
c_args = ['-fno-strict-aliasing', '-fwrapv']
cpp_args = ['-fno-strict-aliasing', '-fwrapv']
